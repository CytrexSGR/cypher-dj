# djk/pruefstand/aufnehmer: JACK-Aufnehmer am Ziel (Scheibe 01; Scheibe 16 baut den Prüfstand aus).
cmake_minimum_required(VERSION 3.28)
project(cypherdj_aufnehmer LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE RelWithDebInfo)
endif()
find_package(PkgConfig REQUIRED)
pkg_check_modules(JACK REQUIRED IMPORTED_TARGET jack)
pkg_check_modules(SNDFILE REQUIRED IMPORTED_TARGET sndfile)
add_executable(cypherdj-aufnehmer aufnehmer.cpp)
target_include_directories(cypherdj-aufnehmer PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/../../kern/include)
target_link_libraries(cypherdj-aufnehmer PRIVATE PkgConfig::JACK PkgConfig::SNDFILE)
target_compile_options(cypherdj-aufnehmer PRIVATE -Wall -Wextra)
